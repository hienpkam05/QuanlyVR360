function makeUniqueSceneId(name) {
  const base = generateId(name);
  const existing = new Set(scenes.map(s => s.id));
  if (!existing.has(base)) return base;
  let n = 2;
  while (existing.has(`${base}_${n}`)) n++;
  return `${base}_${n}`;
}
